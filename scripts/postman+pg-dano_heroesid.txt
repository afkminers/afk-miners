INSERT INTO monster_instances (monster_id, map_key, max_hp, hp)
SELECT id, 'house', COALESCE("healthMax",50), COALESCE("healthMax",50)
FROM monsters_master
ORDER BY id
LIMIT 1
RETURNING id, hp, max_hp;

Consultas de verificação rápidas

Mortos prontos pra renascer:

SELECT id, respawn_at, (now() >= respawn_at) AS due
FROM monster_instances
WHERE state='DEAD'
ORDER BY respawn_at NULLS LAST;


Vivos gerados por spawn:

SELECT id, state, spawn_id, map_key
FROM monster_instances
WHERE state='ALIVE' AND spawn_id IS NOT NULL
ORDER BY created_at DESC
LIMIT 5;


Progresso de skill do herói:

WHERE hero_id = '<HERO_ID>'
