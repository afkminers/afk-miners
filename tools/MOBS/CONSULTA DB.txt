WITH need(col, tbl) AS (
  VALUES
  -- spawns
  ('id','spawns'),('mapKey','spawns'),('monsterKey','spawns'),
  ('x','spawns'),('y','spawns'),('w','spawns'),('h','spawns'),
  ('count','spawns'),('respawnSec','spawns'),

  -- monster_instances
  ('id','monster_instances'),('spawn_id','monster_instances'),
  ('state','monster_instances'),('hp','monster_instances'),('max_hp','monster_instances'),
  ('map_key','monster_instances'),          -- toleramos existir ou não
  ('respawn_at','monster_instances'),
  ('created_at','monster_instances'),('updated_at','monster_instances'),

  -- monsters_master
  ('id','monsters_master'),('key','monsters_master'),('healthMax','monsters_master')
)
SELECT n.tbl AS table, n.col AS column, 
       (SELECT COUNT(*) FROM information_schema.columns c 
         WHERE c.table_schema='public' AND c.table_name=n.tbl AND c.column_name=n.col) > 0 AS exists
FROM need n
ORDER BY table, column
;
